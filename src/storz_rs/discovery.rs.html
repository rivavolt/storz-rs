12/// Scan for Storz & Bickel devices via BLE and return discovered peripherals.
13///
14/// The `timeout` controls how long to scan before returning results.
15pub async fn discover_vaporizers(adapter: &Adapter, timeout: Duration) -> Result<Vec<Peripheral>, StorzError> {
16    info!("Starting BLE scan for Storz & Bickel devices ({timeout:?})…");
17
18    // Ensure adapter is powered on before scanning
19    match adapter.adapter_info().await {
20        Ok(info_str) => debug!("Adapter info: {info_str}"),
21        Err(e) => warn!("Could not read adapter info: {e}"),
22    }
23
24    adapter.start_scan(ScanFilter::default()).await.map_err(|e| {
25        StorzError::Bluetooth(btleplug::Error::Other(
26            format!(
27                "Failed to start BLE scan: {e}\n\n\
28                     Troubleshooting:\n\
29                     1. Ensure Bluetooth is enabled: rfkill unblock bluetooth\n\
30                     2. Ensure adapter is powered on: bluetoothctl power on\n\
31                     3. Try running with elevated permissions (sudo or bluetooth group)\n\
32                     4. Check: bluetoothctl show"
33            )
34            .into(),
35        ))
36    })?;
37
38    time::sleep(timeout).await;
39    adapter.stop_scan().await?;
40
41    let peripherals = adapter.peripherals().await?;
42    let mut found = Vec::new();
43
44    for p in peripherals {
45        if let Ok(Some(props)) = p.properties().await {
46            if let Some(name) = props.local_name.as_ref() {
47                if DEVICE_NAME_PREFIXES.iter().any(|prefix| name.contains(prefix)) {
48                    info!("Found device: {name}");
49                    found.push(p);
50                }
51            }
52        }
53    }
54
55    if found.is_empty() {
56        warn!("No Storz & Bickel devices found during scan");
57    } else {
58        debug!("Discovered {} device(s)", found.len());
59    }
60
61    Ok(found)
62}

64/// Scan until the first matching Storz & Bickel device appears, returning it immediately instead of waiting out the full scan window. `filter` further narrows by substring match on the advertised name or the peripheral address.
65pub async fn discover_first(
66    adapter: &Adapter,
67    timeout: Duration,
68    filter: Option<&str>,
69) -> Result<Peripheral, StorzError> {
70    let matches = |name: &str, address: &str| {
71        DEVICE_NAME_PREFIXES.iter().any(|p| name.contains(p))
72            && filter.is_none_or(|f| {
73                name.to_lowercase().contains(&f.to_lowercase()) || address.to_lowercase() == f.to_lowercase()
74            })
75    };
76
77    let mut events = adapter.events().await?;
78    adapter.start_scan(ScanFilter::default()).await?;
79
80    let check = |id: btleplug::platform::PeripheralId| {
81        let adapter = adapter.clone();
82        async move {
83            let p = adapter.peripheral(&id).await.ok()?;
84            let props = p.properties().await.ok().flatten()?;
85            let name = props.local_name.as_deref().unwrap_or("");
86            if matches(name, &p.address().to_string()) {
87                info!("Found device: {name}");
88                Some(p)
89            } else {
90                None
91            }
92        }
93    };
94
95    let result = time::timeout(timeout, async {
96        // Devices already known to the adapter don't re-emit DeviceDiscovered, so sweep the cache first.
97        for p in adapter.peripherals().await.unwrap_or_default() {
98            if let Some(found) = check(p.id()).await {
99                return Some(found);
100            }
101        }
102        while let Some(event) = events.next().await {
103            if let CentralEvent::DeviceDiscovered(id) | CentralEvent::DeviceUpdated(id) = event {
104                if let Some(found) = check(id).await {
105                    return Some(found);
106                }
107            }
108        }
109        None
110    })
111    .await;
112
113    adapter.stop_scan().await.ok();
114    match result {
115        Ok(Some(p)) => Ok(p),
116        Ok(None) => Err(StorzError::DeviceNotFound),
117        Err(_) => Err(StorzError::DeviceNotFound),
118    }
119}

121/// Obtain the default BLE adapter.
122pub async fn get_adapter() -> Result<Adapter, StorzError> {
123    let manager = Manager::new().await?;
124    let adapters = manager.adapters().await?;
125    adapters.into_iter().next().ok_or(StorzError::DeviceNotFound)
126}

128/// Select a single peripheral from a list.
129///
130/// If only one device is found, returns it immediately.
131/// If multiple are found, prints them and prompts for selection via stdin.
132pub async fn select_peripheral(peripherals: Vec<Peripheral>) -> Result<Peripheral, StorzError> {
133    if peripherals.is_empty() {
134        return Err(StorzError::DeviceNotFound);
135    }
136    if peripherals.len() == 1 {
137        return Ok(peripherals.into_iter().next().unwrap());
138    }
139
140    println!("\nMultiple devices found:");
141    let mut names = Vec::new();
142    for (i, p) in peripherals.iter().enumerate() {
143        let name =
144            p.properties().await.ok().flatten().and_then(|props| props.local_name).unwrap_or_else(|| "Unknown".into());
145        names.push(name.clone());
146        println!("  [{}] {}", i + 1, name);
147    }
148
149    loop {
150        print!("\nSelect device [1-{}]: ", peripherals.len());
151        use std::io::Write;
152        std::io::stdout().flush().ok();
153
154        let mut input = String::new();
155        std::io::stdin().read_line(&mut input).ok();
156        if let Ok(idx) = input.trim().parse::<usize>() {
157            if idx >= 1 && idx <= peripherals.len() {
158                info!("Selected: {}", names[idx - 1]);
159                return Ok(peripherals.into_iter().nth(idx - 1).unwrap());
160            }
161        }
162        println!("Invalid selection. Try again.");
163    }
164}
